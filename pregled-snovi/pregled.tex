\section{Osnove o vektorjih}

\subsection{Definicija}

\noindent
\begin{minipage}[t]{0.58\textwidth}
\vspace{0pt}
Vektor je določen s smerjo, usmerjenostjo in dolžino.

\medskip
\textbf{Enaka vektorja} sta vzporedna, enako dolga in enako usmerjena.

\medskip
\textbf{Nasprotni vektor} $-\vect{a}$ je enako dolg kot $\vect{a}$, vendar usmerjen v nasprotno smer.

\medskip
\textbf{Ničelni vektor} $\vect{0}$ ima dolžino $0$.

\medskip
\textbf{Oznaka:} Vektor označimo z $\vect{AB}$ ali z $\vect{a}$.

\section{Seštevanje in odštevanje vektorjev}

Vektorje lahko seštevamo in odštevamo grafično ali z računanjem komponent.

\subsection{Grafično seštevanje}

Vektorja seštejemo tako, da ju postavimo po vrsti ali uporabimo paralelogramsko pravilo. V obeh primerih je njuna vsota prikazana z rdečim vektorjem.

\medskip
\noindent\begin{minipage}[t]{0.48\textwidth}
\vspace{0pt}
\centering

Začetek $\vect{b}$ postavimo v konec $\vect{a}$. Vsota gre od začetka $\vect{a}$ do konca $\vect{b}$. \vspace{2mm}

\subsection{Lastnosti seštevanja}

\begin{itemize}[itemsep=4pt]
    \item komutativnost: $\vect{a}+\vect{b}=\vect{b}+\vect{a}$,
    \item asociativnost: $(\vect{a}+\vect{b})+\vect{c}=\vect{a}+(\vect{b}+\vect{c})$,
    \item nevtralni element: $\vect{a}+\vect{0}=\vect{a}$,
    \item nasprotni element: $\vect{a}+(-\vect{a})=\vect{0}$.
\end{itemize}

\subsection{Odštevanje}

Odštevanje definiramo kot prištevanje nasprotnega vektorja:
\[
\pomembno{\vect{a}-\vect{b}=\vect{a}+(-\vect{b}).}
\]

\begin{minipage}[c]{0.38\textwidth}

\end{tikzpicture}

Skalar je realno število. Vektor $\vect{a}$, pomnožen s skalarjem $k$, zapišemo kot
\(
\pomembno{k\vect{a}.}
\)


\subsection{Pravila}

\begin{itemize}[itemsep=4pt]
    \item $\pomembno{\abs{k\vect{a}}=|k|\,\abs{\vect{a}}}$,
    \item $k>0$: ista usmerjenost,
    \item $k<0$: nasprotna usmerjenost,
    \item $k=0$: dobimo ničelni vektor.
\end{itemize}

\section{Komponente vektorja}

Vektor lahko zapišemo s komponentami v koordinatnem sistemu. 
\[
\pomembno{\vect{a}=(a_1,a_2,a_3)=a_1\vect{i}+a_2\vect{j}+a_3\vect{k}.}
\]

\subsection{Operacije s komponentami}

\begin{itemize}[itemsep=4pt]
    \item seštevanje: $\pomembno{\vect{a}+\vect{b}=(a_1+b_1,a_2+b_2,a_3+b_3)}$,
    \item odštevanje: $\pomembno{\vect{a}-\vect{b}=(a_1-b_1,a_2-b_2,a_3-b_3)}$,
    \item množenje s skalarjem: $\pomembno{k\vect{a}=(ka_1,ka_2,ka_3)}$.
\end{itemize}


\subsection{\texorpdfstring{\textcolor{barvaNaslovaPrimera}{Primer}}{Primer}}
\begin{primer}
Če je $\vect{a}=(2,3)$ in $\vect{b}=(1,1)$, seštejemo oziroma odštejemo istoležna koordinata:
\[
\vect{a}+\vect{b}=(3,4),
\qquad
\vect{a}-\vect{b}=(1,2).
\]
\end{primer}
\subsection{\texorpdfstring{\textcolor{barvaNaslovaPrimera}{Primer}}{Primer}}
\begin{primer}
Če je $\vect{a}=(2,1,3)$ in $\vect{b}=(4,-1,2)$, seštejemo istoležne komponente; pri množenju vsako komponento pomnožimo z $2$:
\[
\vect{a}+\vect{b}=(6,0,5),
\qquad
2\vect{a}=(4,2,6).
\]
\end{primer}


\subsection{Dolžina vektorja}

Dolžina vektorja $\vect{AB}$ je razdalja med točkama $A$ in $B$:
\[
\pomembno{\abs{\vect{AB}}=d(A,B)=\sqrt{(b_1-a_1)^2+(b_2-a_2)^2+(b_3-a_3)^2}}.
\]

\subsection{\texorpdfstring{\textcolor{barvaNaslovaPrimera}{Primer}}{Primer}}
\begin{primer}
Naj bo $A(1,2)$ in $B(4,5)$. Razliko koordinat kvadriramo, seštejemo in izračunamo koren:
\[
\pomembno{\abs{\vect{AB}}=\sqrt{(4-1)^2+(5-2)^2}=\sqrt{9+9}=3\sqrt{2}.}
\]
\end{primer}
\subsection{Enotski vektor}

Enotski vektor v smeri neničelnega vektorja $\vect{a}$ ima dolžino $1$:
\[
\pomembno{\vect{a}_e=\frac{1}{\abs{\vect{a}}}\vect{a},\qquad \abs{\vect{a}_e}=1.}
\]

\subsection{\texorpdfstring{\textcolor{barvaNaslovaPrimera}{Primer}}{Primer}}
\begin{primer}
Naj bo $\vect{a}=(3,4)$. Njegova dolžina je $5$, zato vsako koordinato delimo s $5$:
\[
\abs{\vect{a}}=5,
\qquad
\vect{a}_e=\left(\frac{3}{5},\frac{4}{5}\right).
\]
\end{primer}




\section{Baza in linearne kombinacije}

\subsection{Linearna kombinacija}

Linearna kombinacija vektorjev $\vect{a}$ in $\vect{b}$ je izraz oblike
\[
\pomembno{\alpha\vect{a}+\beta\vect{b},\qquad \alpha,\beta\in\mathbb{R}.}
\]

\subsection{Baza ravnine}

Dva linearno neodvisna vektorja $\vect{a},\vect{b}$ tvorita bazo ravnine. Vsak vektor v ravnini lahko zapišemo kot
\[
\pomembno{\vect{v}=x\vect{a}+y\vect{b}.}
\]

\subsection{Baza prostora}

Tri linearno neodvisna vektorja v prostoru tvorijo bazo prostora. Vsak vektor v prostoru lahko zapišemo kot
\[
\pomembno{\vect{v}=x\vect{a}+y\vect{b}+z\vect{c}.}
\]

\subsection{\texorpdfstring{\textcolor{barvaNaslovaPrimera}{Primer}}{Primer}}
\begin{primer}
Zapis $\vect{u}=2\vect{a}-3\vect{b}+5\vect{c}$ je linearna kombinacija vektorjev $\vect{a},\vect{b},\vect{c}$; koeficienti so $2$, $-3$ in $5$.
\end{primer}

\section{Odvisni in neodvisni vektorji}

Vektorji so linearno odvisni, če lahko enega zapišemo kot linearno kombinacijo ostalih.

\subsection{Definicija}

Vektorja $\vect{a}$ in $\vect{b}$ sta linearno neodvisna, če iz
\[
\pomembno{\alpha\vect{a}+\beta\vect{b}=\vect{0}}
\]
sledi
\[
\pomembno{\alpha=0\quad\text{in}\quad\beta=0.}
\]
Če obstaja tudi kakšna druga rešitev, sta vektorja linearno odvisna.

\subsection{Geometrijska interpretacija}

\begin{itemize}[itemsep=4pt]
    \item Če sta vektorja vzporedna, sta linearno odvisna.
    \item V ravnini za bazo potrebujemo dva linearno neodvisna vektorja.
\end{itemize}


\subsection{\texorpdfstring{\textcolor{barvaNaslovaPrimera}{Primer}}{Primer}}
\begin{primer}
Vektorja $\vect{a}=(2,3)$ in $\vect{b}=(4,1)$ sta linearno neodvisna, saj enačba
\[
\alpha\vect{a}+\beta\vect{b}=\vect{0}
\]
po koordinatah da sistem
\[
\begin{cases}
2\alpha+4\beta=0,\\
3\alpha+\beta=0.
\end{cases}
\]
Iz druge enačbe sledi $\beta=-3\alpha$. Če to vstavimo v prvo enačbo, dobimo
\[
2\alpha-12\alpha=0
\quad\Rightarrow\quad
\alpha=0.
\]
Zato je tudi $\beta=0$. Edina rešitev je torej $\alpha=\beta=0$, zato sta vektorja linearno neodvisna.
\end{primer}


\section{Skalarni produkt}

Skalarni produkt dveh vektorjev je število. Če je $\varphi$ kot med njima, velja
\[
\pomembno{\vect{a}\cdot\vect{b}=\abs{\vect{a}}\,\abs{\vect{b}}\cos\varphi.}
\]

\subsection{Lastnosti}

\begin{itemize}[itemsep=4pt]
    \item komutativnost: $\vect{a}\cdot\vect{b}=\vect{b}\cdot\vect{a}$,
    \item distributivnost: $\vect{a}\cdot(\vect{b}+\vect{c})=\vect{a}\cdot\vect{b}+\vect{a}\cdot\vect{c}$,
    \item homogenost: $(k\vect{a})\cdot\vect{b}=k(\vect{a}\cdot\vect{b})$.
\end{itemize}

\subsection{Pomen skalarnega produkta}

\begin{itemize}[itemsep=4pt]
    \item Če je $\varphi=90^\circ$, je $\vect{a}\cdot\vect{b}=0$; vektorja sta pravokotna.
    \item Če je $\varphi=0^\circ$, je $\vect{a}\cdot\vect{b}=\abs{\vect{a}}\abs{\vect{b}}$.
    \item Če je $\varphi=180^\circ$, je $\vect{a}\cdot\vect{b}=-\abs{\vect{a}}\abs{\vect{b}}$.
\end{itemize}

\subsection{Dolžina vektorja preko skalarnega produkta}

\[
\pomembno{\abs{\vect{a}}=\sqrt{\vect{a}\cdot\vect{a}}.}
\]

\subsection{\texorpdfstring{\textcolor{barvaNaslovaPrimera}{Primer}}{Primer}}
\begin{primer}
Naj bo $\vect{a}=(3,4)$ in $\vect{b}=(1,2)$. Pomnožimo istoležne koordinate in seštejemo:
\[
\vect{a}\cdot\vect{b}=3\cdot1+4\cdot2=11.
\]
\end{primer}



\section{Skalarni produkt s komponentami}

Če sta vektorja podana s komponentami,
\[
\vect{a}=(a_1,a_2,a_3),\qquad \vect{b}=(b_1,b_2,b_3),
\]
potem je
\[
\pomembno{\vect{a}\cdot\vect{b}=a_1b_1+a_2b_2+a_3b_3.}
\]
To je običajen način računanja skalarnega produkta v koordinatnem sistemu.

\subsection{\texorpdfstring{\textcolor{barvaNaslovaPrimera}{Primer}}{Primer}}
\begin{primer}
Naj bo $\vect{a}=(1,2,3)$ in $\vect{b}=(4,0,-1)$. Pomnožimo istoležne komponente in seštejemo:
\[
\vect{a}\cdot\vect{b}=1\cdot4+2\cdot0+3\cdot(-1)=4-3=1.
\]
\end{primer}

\subsection{Povezava z dolžino}

Iz formule za skalarni produkt s komponentami dobimo
\[
\pomembno{\abs{\vect{a}}=\sqrt{a_1^2+a_2^2+a_3^2}.}
\]

\section{Povzetek}

Najpomembnejše lastnosti vektorjev so:
\begin{itemize}[itemsep=4pt]
    \item vektor ima smer, usmerjenost in dolžino,
    \item vektorje seštevamo in odštevamo po pravilih za vektorje,
    \item množenje s skalarjem spremeni dolžino in lahko obrne usmerjenost,
    \item linearno neodvisni vektorji tvorijo bazo,
    \item skalarni produkt je povezan z dolžinama vektorjev in kotom med njima,
    \item s komponentami je računanje z vektorji preprostejše.
\end{itemize}
