\naloga{1. naloga}{11}
V paralelogramu $ABCD$ označimo $\vec a=\vec{AB}$ in $\vec b=\vec{AD}$. Zapiši zahtevane vektorje s pomočjo $\vec a$ in/ali $\vec b$.
\begin{enumerate}[label=\alph*.]
    \item $\vec{AB}+\vec{BC}$ \hfill (5)
    \resitev{Ker je v paralelogramu $\vec{BC}=\vec{AD}=\vec b$, dobimo \koncni{$\vec{AB}+\vec{BC}=\vec a+\vec b=\vec{AC}$.}}
    \item $\vec{BC}+\vec{CD}+\vec{DA}$ \hfill (6)
    \resitev{$\vec{BC}=\vec b$, $\vec{CD}=-\vec a$ in $\vec{DA}=-\vec b$, zato je \koncni{$\vec{BC}+\vec{CD}+\vec{DA}=-\vec a$}.}
\end{enumerate}

\naloga{2. naloga}{14}
Dana sta vektorja $\vec a=(2,-1,3)$ in $\vec b=(-1,4,2)$.
\begin{enumerate}[label=\alph*.]
    \item Izračunaj $\vec a-\vec b$. \hfill (7)
    \resitev{$\vec a-\vec b=(2-(-1),-1-4,3-2)=\koncni{(3,-5,1)}$.}
    \item Izračunaj $2\vec a-\dfrac12\vec b$. \hfill (7)
    \resitev{$2\vec a=(4,-2,6)$ in $\frac12\vec b=(-\frac12,2,1)$, zato je \koncni{$2\vec a-\frac12\vec b=(\frac92,-4,5)$}.}
\end{enumerate}

\naloga{3. naloga}{17}
Podani sta točki $A(-1,2,0)$ in $B(3,-1,2)$.
\begin{enumerate}[label=\alph*.]
    \item Določi komponente vektorja $\vec{AB}$. \hfill (5)
    \resitev{$\vec{AB}=B-A=(3-(-1),-1-2,2-0)=\koncni{(4,-3,2)}$.}
    \item Izračunaj dolžino vektorja $\vec{AB}$. \hfill (6)
    \resitev{$|\vec{AB}|=\sqrt{4^2+(-3)^2+2^2}=\sqrt{16+9+4}=\koncni{\sqrt{29}}$.}
    \item Zapiši enotski vektor v smeri $\vec{AB}$. \hfill (6)
    \resitev{$\vec{e}_{AB}=\frac{\vec{AB}}{|\vec{AB}|}=\koncni{\left(\frac4{\sqrt{29}},-\frac3{\sqrt{29}},\frac2{\sqrt{29}}\right)}$.}
\end{enumerate}

\naloga{4. naloga}{14}
Dana sta vektorja $\vec p=(1,2,-2)$ in $\vec q=(2,-1,0)$.
\begin{enumerate}[label=\alph*.]
    \item Izračunaj skalarni produkt $\vec p\cdot\vec q$. \hfill (5)
    \resitev{$\vec p\cdot\vec q=1\cdot2+2\cdot(-1)+(-2)\cdot0=\koncni{0}$.}
    \item Izračunaj $|\vec q|$. \hfill (4)
    \resitev{$|\vec q|=\sqrt{2^2+(-1)^2+0^2}=\koncni{\sqrt5}$.}
    \item Utemelji, ali sta $\vec p$ in $\vec q$ pravokotna. \hfill (5)
    \resitev{Ker je njun skalarni produkt enak $0$, je \koncni{da, vektorja sta pravokotna}.}
\end{enumerate}

\naloga{5. naloga}{8}
Podana sta vektorja $\vec u=(1,1,0)$ in $\vec v=(1,0,1)$. Izračunaj kot $\varphi$ med vektorjema.
\resitev{Uporabimo $\cos\varphi=\frac{\vec u\cdot\vec v}{|\vec u|\,|\vec v|}$. Ker je $\vec u\cdot\vec v=1$ ter $|\vec u|=|\vec v|=\sqrt2$, je $\cos\varphi=\frac12$, zato je \koncni{$\varphi=60^\circ$}.}

\naloga{6. naloga}{16}
\begin{enumerate}[label=\alph*.]
    \item Dana sta $\vec u=2\vec a-\vec b$ in $\vec v=\vec a+3\vec b$. Izrazi $2\vec u-\vec v$ kot linearno kombinacijo vektorjev $\vec a$ in $\vec b$. \hfill (7)
    \resitev{$2\vec u-\vec v=2(2\vec a-\vec b)-(\vec a+3\vec b)=4\vec a-2\vec b-\vec a-3\vec b=\koncni{3\vec a-5\vec b}$.}
    \item Reši vektorsko enačbo $2(\vec x+\vec a)=3\vec b+\vec x$ in izrazi $\vec x$ z vektorjema $\vec a$ in $\vec b$. \hfill (9)
    \resitev{$2\vec x+2\vec a=3\vec b+\vec x$. Odštejemo $\vec x$ in $2\vec a$ ter dobimo \koncni{$\vec x=3\vec b-2\vec a$}.}
\end{enumerate}

\naloga{7. naloga}{20}
Podani so vektorji $\vec a=(1,2)$, $\vec b=(3,6)$ in $\vec c=(-2,1)$.
\begin{enumerate}[label=\alph*.]
    \item Pokaži, da sta $\vec a$ in $\vec b$ linearno odvisna. \hfill (5)
    \resitev{$\vec b=(3,6)=3(1,2)=3\vec a$, zato sta \koncni{vektorja linearno odvisna}.}
    \item Presodi, ali sta $\vec a$ in $\vec c$ linearno neodvisna. Odgovor utemelji. \hfill (5)
    \resitev{Determinanta je $\begin{vmatrix}1&-2\\2&1\end{vmatrix}=1+4=5\ne0$, zato sta \koncni{vektorja linearno neodvisna}.}
    \item Kateri od parov $\{\vec a,\vec b\}$ in $\{\vec a,\vec c\}$ tvori bazo ravnine? Utemelji izbiro. \hfill (5)
    \resitev{Ker je $\{\vec a,\vec c\}$ neodvisen par, medtem ko je $\{\vec a,\vec b\}$ odvisen, \koncni{bazo ravnine tvori par $\{\vec a,\vec c\}$}.}
    \item Ali lahko trije vektorji v ravnini tvorijo bazo ravnine? Kratko pojasni. \hfill (5)
    \resitev{Baza ravnine vsebuje natanko dva linearno neodvisna vektorja; zato \koncni{trije vektorji v ravnini ne morejo tvoriti baze ravnine}.}
\end{enumerate}
